# Pistas Conceptuales

- Archivos ocultos: ls -la  (empiezan con punto)
- grep: grep -r "PATRON" directorio/
- find: find . -name "archivo.txt"
- Permisos: ls -l  (rw-r--r-- = 644)
- Logs: grep "PALABRA" archivo.log
- tar: tar -xzf archivo.tar.gz
- Renombrar: mv viejo.enc nuevo.txt
- Pipeline: comando1 | comando2 | comando3
- Base64: base64 -d archivo.b64
